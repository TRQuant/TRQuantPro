```python
def cn_today_str_compact():
    """获取中国时间（UTC+8）的当前日期字符串（紧凑格式YYYYMMDD，用于AKShare接口）"""
    cn_now = datetime.now(timezone.utc) + timedelta(hours=8)
    return cn_now.strftime('%Y%m%d')
```
